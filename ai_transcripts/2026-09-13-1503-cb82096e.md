def bateman(t, ka=4.0, ke=0.9):
    """One dose's caffeine level over time: absorption then first-order elimination."""

def curve_points(x0=190, x1=850, base=700, height=400, t_end=3.6, n=160):

def bateman(t, ka, ke):
    """One dose's caffeine level over time: absorption then first-order elimination."""
    return math.exp(-ke * t) - math.exp(-ka * t)

    peak = max(bateman(i / 4000 * t_end, ka, ke) for i in range(1, 4001))
    pts = []
    for i in range(n + 1):
        t = i / n * t_end
        pts.append((x0 + (x1 - x0) * i / n, base - height * bateman(t, ka, ke) / peak))
    return pts

    """The point on the falling side where the level is half the peak."""
    peak_i = min(range(len(pts)), key=lambda i: pts[i][1])
    target = base - height / 2
    for i in range(peak_i, len(pts)):
        if pts[i][1] >= target:
            return pts[i]
    return pts[-1]

def outline(pts, width):
    """The stroke of a polyline as a filled outline with round caps, so no renderer can close the open path."""
    r = width / 2
    left, right = [], []
    for i, (x, y) in enumerate(pts):
        ax, ay = pts[max(i - 1, 0)]
        bx, by = pts[min(i + 1, len(pts) - 1)]
        tx, ty = bx - ax, by - ay
        length = math.hypot(tx, ty)
        tx, ty = tx / length, ty / length
        nx, ny = -ty, tx
        left.append((x + nx * r, y + ny * r))
        right.append((x - nx * r, y - ny * r))

        x, y = pts[i]
        j = i + (1 if i == 0 else -1)
        tx, ty = (x - pts[j][0]), (y - pts[j][1])
        length = math.hypot(tx, ty)
        tx, ty = tx / length, ty / length

"""Scratch generator: writes Icon Composer (.icon) concepts for Half-Life and renders them with ictool."""

import json
import math
import os
import subprocess
import sys

HERE = os.path.dirname(os.path.abspath(__file__))
ICTOOL = "/Applications/Xcode.app/Contents/Applications/Icon Composer.app/Contents/Executables/ictool"
RENDITIONS = ("Default", "Dark", "TintedLight", "TintedDark", "ClearLight", "ClearDark")


def hexrgb(h):
    h = h.lstrip("#")
    return tuple(int(h[i:i + 2], 16) / 255 for i in (0, 2, 4))


def srgb(h, a=1.0):
    r, g, b = hexrgb(h)
    return "srgb:%.5f,%.5f,%.5f,%.5f" % (r, g, b, a)


def vertical(top, bottom):
    return {"linear-gradient": [srgb(top), srgb(bottom)],
            "orientation": {"start": {"x": 0.5, "y": 0}, "stop": {"x": 0.5, "y": 1}}}


def bateman(t, ka, ke):
    """One dose's caffeine level over time: absorption then first-order elimination."""
    return math.exp(-ke * t) - math.exp(-ka * t)


def curve_points(x0, x1, base, height, ka, ke, t_end, n=240):
    peak = max(bateman(i / 4000 * t_end, ka, ke) for i in range(1, 4001))
    return [(x0 + (x1 - x0) * i / n, base - height * bateman(i / n * t_end, ka, ke) / peak) for i in range(n + 1)]


def half_life_point(pts, base, height):
    """The point on the falling side where the level is half the peak."""
    peak_i = min(range(len(pts)), key=lambda i: pts[i][1])
    for i in range(peak_i, len(pts)):
        if pts[i][1] >= base - height / 2:
            return pts[i]
    return pts[-1]


def fmt(pts):
    return " L ".join("%.1f %.1f" % p for p in pts)


def outline(pts, width):
    """The stroke of a polyline as a filled outline with round caps, so no renderer can close the open path."""
    r = width / 2
    left, right = [], []
    for i, (x, y) in enumerate(pts):
        ax, ay = pts[max(i - 1, 0)]
        bx, by = pts[min(i + 1, len(pts) - 1)]
        length = math.hypot(bx - ax, by - ay)
        nx, ny = -(by - ay) / length, (bx - ax) / length
        left.append((x + nx * r, y + ny * r))
        right.append((x - nx * r, y - ny * r))

    def cap(i):
        x, y = pts[i]
        j = i + (1 if i == 0 else -1)
        length = math.hypot(x - pts[j][0], y - pts[j][1])
        tx, ty = (x - pts[j][0]) / length, (y - pts[j][1]) / length
        nx, ny = -ty, tx
        return [(x + r * (nx * math.cos(a) + tx * math.sin(a)), y + r * (ny * math.cos(a) + ty * math.sin(a)))
                for a in (k / 24 * math.pi for k in range(1, 24))]

    return "M " + fmt(left + cap(len(pts) - 1) + right[::-1] + cap(0)) + " Z"


def svg(body):
    return ('<svg xmlns="http://www.w3.org/2000/svg" width="1024" height="1024" viewBox="0 0 1024 1024">'
            + body + "</svg>\n")


def write_icon(root, fill, groups, dark_fill=None):
    assets = os.path.join(root, "Assets")
    os.makedirs(assets, exist_ok=True)
    doc_groups = []
    for group in groups:
        layers = []
        for layer in group["layers"]:
            with open(os.path.join(assets, layer["file"]), "w") as f:
                f.write(svg(layer["svg"]))
            layers.append({"glass": True, "image-name": layer["file"], "name": layer["file"][:-4]})
        doc_groups.append({
            "layers": layers,
            "name": group["name"],
            "shadow": {"kind": "neutral", "opacity": 0.5},
            "translucency": {"enabled": group.get("translucent", True), "value": 0.4},
        })
    doc = {"groups": doc_groups, "supported-platforms": {"circles": ["watchOS"], "squares": "shared"}}
    if dark_fill is None:
        doc["fill"] = fill
    else:
        doc["fill-specializations"] = [{"value": fill}, {"appearance": "dark", "value": dark_fill}]
    with open(os.path.join(root, "icon.json"), "w") as f:
        json.dump(doc, f, indent=2, sort_keys=True)
        f.write("\n")
    return root


def render(icon, rendition, out_dir, size=1024):
    out = os.path.join(out_dir, "%s-%s.png" % (os.path.basename(icon)[:-5], rendition))
    os.makedirs(out_dir, exist_ok=True)
    cmd = [ICTOOL, icon, "--export-image", "--output-file", out, "--platform", "iOS",
           "--rendition", rendition, "--width", str(size), "--height", str(size), "--scale", "1"]
    if rendition.startswith("Tinted"):
        cmd += ["--tint-color", "0.08", "--tint-strength", "0.8"]
    r = subprocess.run(cmd, capture_output=True, text=True)
    if r.returncode != 0 or not os.path.exists(out):
        print("FAILED", rendition, os.path.basename(icon), r.returncode, r.stderr.strip())


COPPER_LIGHT, CREAM = "#E6B173", "#F6EFE3"
ESPRESSO_FILL = vertical("#3D342B", "#1B1611")


def design(base, height, x0, x1, ka, ke, t_end, stroke, dot_r, moon_c, moon_r):
    """Moon + dot: one dose's decay curve with a soft fill, a cream dot where half of it is gone,
    and a crescent moon for bedtime above the tail."""
    pts = curve_points(x0=x0, x1=x1, base=base, height=height, ka=ka, ke=ke, t_end=t_end)
    hx, hy = half_life_point(pts, base, height)
    area = "M " + fmt(pts) + " L %.1f %.1f L %.1f %.1f Z" % (pts[-1][0], base, pts[0][0], base)
    mx, my = moon_c
    moon = ('<mask id="bite"><rect width="1024" height="1024" fill="white"/>'
            '<circle cx="%.1f" cy="%.1f" r="%.1f" fill="black"/></mask>'
            '<circle cx="%.1f" cy="%.1f" r="%.1f" fill="%s" mask="url(#bite)"/>'
            % (mx + moon_r * 0.42, my - moon_r * 0.34, moon_r * 0.86, mx, my, moon_r, CREAM))
    return [
        {"name": "Curve", "layers": [
            {"file": "area.svg", "svg": '<path d="%s" fill="%s" fill-opacity="0.26"/>' % (area, COPPER_LIGHT)},
            {"file": "curve.svg", "svg": '<path d="%s" fill="%s"/>' % (outline(pts, stroke), COPPER_LIGHT)},
        ]},
        {"name": "Markers", "translucent": False, "layers": [
            {"file": "dot.svg", "svg": '<circle cx="%.1f" cy="%.1f" r="%d" fill="%s"/>' % (hx, hy, dot_r, CREAM)},
            {"file": "moon.svg", "svg": moon},
        ]},
    ]


VARIANTS = {
    # a: round 2's layout, a gentler rise, the moon upper right.
    "a": dict(base=700, height=350, x0=200, x1=840, ka=1.7, ke=0.55, t_end=5.0, stroke=60, dot_r=50,
              moon_c=(770, 270), moon_r=86),
    # b: a wider, lower curve and a smaller moon, so the dot and moon don't compete.
    "b": dict(base=730, height=380, x0=170, x1=860, ka=1.7, ke=0.5, t_end=5.2, stroke=62, dot_r=48,
              moon_c=(780, 250), moon_r=72),
}

if __name__ == "__main__":
    out_dir = sys.argv[1]
    size = int(sys.argv[2]) if len(sys.argv) > 2 else 1024
    names = sys.argv[3:] or list(VARIANTS)
    for name in names:
        icon = write_icon(os.path.join(HERE, "moon-dot-%s.icon" % name), ESPRESSO_FILL, design(**VARIANTS[name]))
        for rendition in RENDITIONS:
            render(icon, rendition, out_dir, size)
